# Fireball Deflect: the player stands facing south (yaw 0) at 5000.5 -60 5000.5.
# A: a fireball from 12 blocks south, at eye height, coming straight in: hit back (it ends up south of the player,
#    moving away, or the Pos line errors if it hit the player and exploded).
# B: one from 10 blocks west (90 degrees off the view): flick, hit back (ends up west), aim turned back to about 0 0.
# C: one passing 4 blocks to the side: left alone (ends up east, rotation stays exactly 0 0).
# Setup: python tools/testcfg.py setup fireball-deflect -- criticals enabled=false
# Run:   sh tools/run-test.sh tools/examples/fireball-deflect.txt
# Check: the fireball Pos and player Rotation lines; look at the screenshots.
# Restore: python tools/testcfg.py restore
title 40
world
cmd gamerule send_command_feedback true
cmd gamerule advance_time false
cmd gamerule spawn_mobs false
cmd gamerule mob_griefing false
cmd time set 6000
cmd kill @e[type=!player]
move 800 450
cmd tp @s 5000.5 -60 5000.5 0 0
wait 60
# A: head-on
cmd summon fireball 5000.5 -58.6 5012.5 {Motion:[0.0d,0.0d,-0.5d],acceleration_power:0.1d,ExplosionPower:1b}
wait 40
shot a
cmd data get entity @e[type=fireball,limit=1] Pos
cmd data get entity @s Rotation
cmd kill @e[type=fireball]
cmd tp @s 5000.5 -60 5000.5 0 0
wait 20
# B: from the west
cmd summon fireball 4990.5 -58.6 5000.5 {Motion:[0.5d,0.0d,0.0d],acceleration_power:0.1d,ExplosionPower:1b}
wait 40
shot b
cmd data get entity @e[type=fireball,limit=1] Pos
cmd data get entity @s Rotation
cmd kill @e[type=fireball]
cmd tp @s 5000.5 -60 5000.5 0 0
wait 20
# C: passing 4 blocks to the south
cmd summon fireball 4990.5 -58.6 5004.5 {Motion:[0.5d,0.0d,0.0d],acceleration_power:0.1d,ExplosionPower:1b}
wait 40
shot c
cmd data get entity @e[type=fireball,limit=1] Pos
cmd data get entity @s Rotation
wait 5
quit
