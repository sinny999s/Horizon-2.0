# Aura dynamic-switch: two zombies (stand-ins for players; the targeting is the same apart from the teammate check).
# 1. Aura (LEGIT, sort Distance) locks onto A, dead ahead; B stands 45 degrees to the right, untouched.
# 2. The mouse turns 32 degrees towards B (213 px at sensitivity 0.5): with dynamic-switch, Aura goes to B.
# 3. C is A's twin far ahead; once Aura holds C, D appears right in front on the same line: Aura switches to D.
# Setup: python tools/testcfg.py setup aura rotation-mode=LEGIT sort=Distance range=3.2 mobs=true hostiles=true only-angry-hostiles=false dynamic-switch=true combat-mode=COOLDOWN reaction-min=35 reaction-max=85
# Run:   sh tools/run-test.sh tools/examples/aura-switch.txt
# Check: the Health lines (max 200). On: A drops then stops, B drops after the flick; D drops, C stops.
#        Control run with dynamic-switch=false: B and D stay at 200.
# Restore: python tools/testcfg.py restore
title 40
world
cmd gamerule send_command_feedback true
cmd gamerule advance_time false
cmd gamerule spawn_mobs false
cmd time set 6000
cmd kill @e[type=!player]
move 800 450
cmd tp @s 3000.5 -60 3000.5 0 0
wait 60
cmd item replace entity @s weapon.mainhand with minecraft:diamond_sword
cmd summon zombie 3000.5 -60 3002.8 {NoAI:1b,Silent:1b,PersistenceRequired:1b,Tags:["A"],Health:200f,attributes:[{id:"minecraft:max_health",base:200}]}
cmd summon zombie 2998.7 -60 3002.3 {NoAI:1b,Silent:1b,PersistenceRequired:1b,Tags:["B"],Health:200f,attributes:[{id:"minecraft:max_health",base:200}]}
wait 60
shot locked
cmd data get entity @e[tag=A,limit=1] Health
cmd data get entity @e[tag=B,limit=1] Health
move 1013 450
wait 60
shot flicked
cmd data get entity @e[tag=A,limit=1] Health
cmd data get entity @e[tag=B,limit=1] Health
cmd kill @e[type=zombie]
cmd tp @s 3000.5 -60 3000.5 0 0
wait 20
cmd summon zombie 3000.5 -60 3003.4 {NoAI:1b,Silent:1b,PersistenceRequired:1b,Tags:["C"],Health:200f,attributes:[{id:"minecraft:max_health",base:200}]}
wait 40
cmd data get entity @e[tag=C,limit=1] Health
cmd summon zombie 3000.5 -60 3001.9 {NoAI:1b,Silent:1b,PersistenceRequired:1b,Tags:["D"],Health:200f,attributes:[{id:"minecraft:max_health",base:200}]}
wait 60
shot blocked
cmd data get entity @e[tag=C,limit=1] Health
cmd data get entity @e[tag=D,limit=1] Health
wait 5
quit
