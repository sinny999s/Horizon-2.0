cmd gamerule send_command_feedback true
cmd time set noon
cmd weather clear
cmd tp @s 2000.5 -60 2000.5 0 0
wait 60
cmd kill @e[type=!player]

# 1. Place stash cluster fully inside chunk 125, 125 (X: 2000..2015, Z: 2000..2015)

# 2. Place a single control chest in chunk 127 (X: 2035) -> should NOT trigger StashFinder

# Verify StashFinder detected the stash cluster and renders box, label, and tracers

shot 1_stash_finder_alert_and_box
