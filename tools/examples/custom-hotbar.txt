# ShotBot test for CustomHotbar (Horizon glass dock, smooth selector, offhand, XP, health & food pills)
title 40
world
wait 50

cmd gamerule spawn_mobs false
cmd gamerule send_command_feedback true
cmd time set noon
cmd weather clear
cmd tp @s 2000.5 -60 2000.5 0 0
wait 50
cmd kill @e[type=!player]

cmd gamemode survival
cmd item replace entity @s hotbar.0 with minecraft:netherite_sword
cmd item replace entity @s hotbar.1 with minecraft:enchanted_golden_apple 64
cmd item replace entity @s hotbar.2 with minecraft:totem_of_undying
cmd item replace entity @s hotbar.3 with minecraft:ender_pearl 16
cmd item replace entity @s hotbar.4 with minecraft:firework_rocket 64
cmd item replace entity @s hotbar.5 with minecraft:clock
cmd item replace entity @s hotbar.8 with minecraft:cooked_beef 64
cmd item replace entity @s weapon.offhand with minecraft:shield
cmd item replace entity @s armor.chest with minecraft:diamond_chestplate
cmd experience add @s 42 levels
cmd effect give @s absorption 120 1
wait 20

# 1. Screenshot slot 0 selected (sword)
shot 1_custom_hotbar_slot0

# 2. Switch to slot 2 (totem, KEY_3 = 32)
key 32
wait 20
shot 2_custom_hotbar_slot2

# 3. Switch to slot 5 (clock, KEY_6 = 35)
key 35
wait 20
shot 3_custom_hotbar_slot5

wait 15
quit
