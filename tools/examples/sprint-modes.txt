# Sprint modes (VANILLA vs OMNI) verification.
# Setup:
#   python tools/testcfg.py setup UI 'bind={"key":12,"hold":false,"mouse":false}' -- sprint enabled=true mode=VANILLA
title 40
world
wait 40
cmd gamerule spawn_mobs false
cmd gamerule advance_time false
cmd time set 6000
cmd kill @e[type=!player]
cmd tp @s 2000.5 -60.0 2000.5 0 0
wait 60

# 1. Baseline with Sprint OFF: 20 ticks of W walks ~4.0 blocks (< 4.75).
toggle sprint off
cmd tp @s 2000.5 -60.0 2000.5 0 0
wait 15
hold keyUp 20
wait 24
cmd execute positioned 2000.5 -60.0 2000.5 if entity @s[distance=3.3..4.75]

# 2. Sprint ON (VANILLA): 20 ticks of W sprints ~5.3 blocks (> 4.9).
toggle sprint on
set sprint mode VANILLA
cmd tp @s 2000.5 -60.0 2000.5 0 0
wait 15
hold keyUp 20
wait 16
shot sprint-vanilla-forward
wait 8
cmd execute positioned 2000.5 -60.0 2000.5 if entity @s[distance=4.9..6.8]

# 3. VANILLA edge case: W + Shift (sneaking) must NOT sprint (< 2.2 blocks in 20 ticks).
cmd tp @s 2000.5 -60.0 2000.5 0 0
wait 15
hold keyShift 22
hold keyUp 20
wait 24
cmd execute positioned 2000.5 -60.0 2000.5 if entity @s[distance=..2.2]

# 4. VANILLA edge case: S (backwards) walks at normal speed without omni-sprint FOV zoom.
cmd tp @s 2000.5 -60.0 2000.5 0 0
wait 15
hold keyDown 20
wait 16
shot sprint-vanilla-backward
wait 8
cmd execute positioned 2000.5 -60.0 2000.5 if entity @s[distance=3.3..4.75]

# 5. Sprint ON (OMNI): forward sprints (> 4.9 blocks) and backward applies original omni TickEvent sprint.
set sprint mode OMNI
cmd tp @s 2000.5 -60.0 2000.5 0 0
wait 15
hold keyUp 20
wait 24
cmd execute positioned 2000.5 -60.0 2000.5 if entity @s[distance=4.9..6.8]

cmd tp @s 2000.5 -60.0 2000.5 0 0
wait 15
hold keyDown 20
wait 16
shot sprint-omni-backward
wait 10

# 6. ClickGUI verification for Sprint (Omni & Vanilla modes).
key I
wait 25
type sprint
wait 10
key RETURN
wait 15
shot gui-sprint-omni
set sprint mode VANILLA
wait 10
shot gui-sprint-vanilla
key ESCAPE
wait 10
quit
