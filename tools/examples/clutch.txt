# Clutch: a 1-wide bridge 30 blocks up (creative, so a miss lands on the floor instead of dying).
# A/B: jump while strafing off the right, then the left side (a knock-up stand-in): Clutch places a block beside the
#      bridge and the player ends on it, at y -29.
# C:   hop off a ledge 3 blocks above the floor (less than min-drop 4): no block, the player lands on the grass.
# D:   jump in place on the bridge: no block.
# Setup: python tools/testcfg.py setup clutch
# Run:   sh tools/run-test.sh tools/examples/clutch.txt
# Check: the Pos lines (A/B y -29, C -60, D -29); after A and B the four cleanup fills (everything around the
#        bridge but the bridge) add up to 1 block, after D to 0; C's own fill says 5 (the ledge). Look at the screenshots.
# Restore: python tools/testcfg.py restore
title 40
world
cmd gamerule send_command_feedback true
cmd gamerule advance_time false
cmd gamerule spawn_mobs false
cmd time set 6000
cmd kill @e[type=!player]
cmd tp @s 4000.5 -29 4002.5 180 0
wait 60
cmd fill 4000 -30 3990 4000 -30 4010 stone
cmd give @s stone 64
wait 20
# A: off the right (east) side
cmd tp @s 4000.5 -29 4002.5 180 0
wait 20
keydown D
keydown SPACE
wait 2
keyup SPACE
wait 5
keyup D
wait 40
shot a
cmd data get entity @s Pos
cmd fill 3995 -35 3985 3999 -25 4015 air replace stone
cmd fill 4001 -35 3985 4005 -25 4015 air replace stone
cmd fill 4000 -35 3985 4000 -31 4015 air replace stone
cmd fill 4000 -29 3985 4000 -25 4015 air replace stone
# B: off the left (west) side
cmd tp @s 4000.5 -29 3996.5 180 0
wait 20
keydown A
keydown SPACE
wait 2
keyup SPACE
wait 5
keyup A
wait 40
shot b
cmd data get entity @s Pos
cmd fill 3995 -35 3985 3999 -25 4015 air replace stone
cmd fill 4001 -35 3985 4005 -25 4015 air replace stone
cmd fill 4000 -35 3985 4000 -31 4015 air replace stone
cmd fill 4000 -29 3985 4000 -25 4015 air replace stone
# C: a ledge 3 blocks above the grass (top at -57)
cmd fill 4020 -58 4000 4020 -58 4004 stone
cmd tp @s 4020.5 -57 4002.5 180 0
wait 20
keydown D
keydown SPACE
wait 2
keyup SPACE
wait 5
keyup D
wait 40
shot c
cmd data get entity @s Pos
cmd fill 4021 -61 3998 4023 -56 4006 air replace stone
# D: jump in place on the bridge
cmd tp @s 4000.5 -29 4006.5 180 0
wait 20
key SPACE
wait 30
shot d
cmd data get entity @s Pos
cmd fill 3995 -35 3985 3999 -25 4015 air replace stone
cmd fill 4001 -35 3985 4005 -25 4015 air replace stone
cmd fill 4000 -35 3985 4000 -31 4015 air replace stone
cmd fill 4000 -29 3985 4000 -25 4015 air replace stone
wait 5
quit
