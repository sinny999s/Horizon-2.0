# Cape: Minecraft's capes on you, fetched from Mojang's texture server: from behind with a few of them, then the same one on
# an elytra, without elytra, and last every cape in turn so each is fetched (tools/test-out/cape.log must have no "Couldn't
# fetch", and the capes folder must end up with 50 pictures).
# Setup:  python tools/testcfg.py setup cape style=MIGRATOR elytra=true
# Run:    sh tools/run-test.sh tools/examples/cape.txt
# Restore: python tools/testcfg.py restore
title 40
world
cmd gamerule spawn_mobs false
cmd gamerule advance_time false
cmd time set 6000
cmd kill @e[type=!player]
cmd tp @s 2000.5 -60 2000.5 0 10
wait 60
cmd gamemode survival
key F5
wait 80
shot migrator
set cape style MINECON_2011
wait 60
shot minecon-2011
set cape style MOJANG_STUDIOS
wait 60
shot mojang-studios
set cape style ANNIVERSARY_15TH
wait 60
shot anniversary
# on an elytra, then with elytra off
cmd item replace entity @s armor.chest with elytra
wait 30
shot elytra-on
set cape elytra false
wait 30
shot elytra-off
set cape elytra true
cmd item replace entity @s armor.chest with air
# every cape
capes-all
wait 300
capes-count
quit
