# GodBridge must not act where it can't bridge: walking backwards on solid ground, then off a 1-block ledge while
# looking up (the crosshair never reaches the edge). Expect no jumps and no stone placed in either part.
# Setup: python tools/testcfg.py setup godbridge activation-blocks=0 auto-angle=NONE pitch-check=false auto-jump=true jump-interval=7 block-selection=NONE
# Run:   sh tools/run-test.sh tools/examples/godbridge-nojump.txt
# Check: both jump counts are 0 and both fills say "No blocks were filled"; the player ends on the lower floor.
# Restore: python tools/testcfg.py restore
title 40
world
cmd gamerule send_command_feedback true
cmd gamerule advance_time false
cmd gamerule spawn_mobs false
cmd time set 6000
cmd kill @e[type=!player]
cmd scoreboard objectives add jumps minecraft.custom:minecraft.jump
# Part 1: a wide floor of dirt, walking back at the bridging angle.
cmd tp @s 3000.5 -60 3010.5 0 79
wait 60
cmd give @s stone 64
cmd scoreboard players set @s jumps 0
wait 20
keydown S
wait 80
keyup S
wait 10
cmd scoreboard players get @s jumps
cmd fill 2990 -61 2980 3010 -50 3020 air replace stone
# Part 2: a dirt step 1 block up, walking back off its edge while looking up.
cmd fill 2995 -61 3040 3005 -61 3060 dirt
cmd fill 2995 -60 3050 3005 -60 3060 dirt
cmd tp @s 3000.5 -59 3054.5 0 -20
wait 40
cmd scoreboard players set @s jumps 0
keydown S
wait 60
keyup S
wait 10
cmd scoreboard players get @s jumps
cmd fill 2990 -61 3030 3010 -50 3070 air replace stone
cmd data get entity @s Pos
shot ledge
wait 5
quit
